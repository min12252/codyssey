# 볼륨 생성 (Docker가 관리하는 저장 공간)
haru@MacBook-Air ~/dev % docker volume create mydata
mydata

# 컨테이너 생성, 볼륨 연결
haru@MacBook-Air ~/dev % docker run -it --name vol-test -v mydata:/data ubuntu
root@5b2d1d329bf2:/# echo "hello volume" > /data/memo.txt
root@5b2d1d329bf2:/# exit
exit

# 컨테이너 다시 확인
haru@MacBook-Air ~/dev % docker start vol-test
vol-test
haru@MacBook-Air ~/dev % docker ps
CONTAINER ID   IMAGE          COMMAND                   CREATED          STATUS         PORTS                                     NAMES
